#树 
树（由数组实现存储）的四种遍历方式：前、中、后序遍历和层序遍历
```cpp
#include<bits/stdc++.h>

using namespace std;

const int maxn = 1e3 + 10;

int ls[maxn], rs[maxn];

//前序遍历 
void dfs1(int x)
{
	cout << x << ' ';
	if(ls[x]) dfs1(ls[x]);
	if(ls[x]) dfs1(rs[x]);
}

// 中序遍历 
void dfs2(int x)
{
	if(ls[x]) dfs2(rs[x]);
	cout << x << ' ';
	if(rs[x]) dfs2(rs[x]);
}

// 后序遍历 
void dfs3(int x)
{
	if(ls[x]) dfs3(ls[x]);
	if(rs[x]) dfs3(rs[x]);
	cout << x << ' ';
}

//层序遍历
void bfs()
{
	queue<int> q;
	q.push(1);
	while(q.size())
	{
		int x = q.front(); 
		q.pop();
		cout << x << ' ';
		if(ls[x]) q.push(ls[x]);
		if(rs[x]) q.push(rs[x]);
	}
} 

int main()
{
	int n; cin >> n;
	for(int i = 1; i <= n; i++) cin >> ls[i] >> rs[i];
	
	dfs1(1);
	cout << '\n';
	dfs2(1);
	cout << '\n';
	dfs3(1);
	cout << '\n';
	bfs();
		
	return 0;
}
```




#深度优先搜索顺序（Depth-FirstSearchOrder/DFS序）
[树中颜色 - 蓝桥云课](https://www.lanqiao.cn/courses/44493/learning/?id=3143548&compatibility=false)
一天，小蓝在他的小屋外发现了一颗神奇的树，它的树叶上闪烁着各种不同的颜色。小蓝对这颗树产生了浓厚的兴趣，于是他决定研究这颗树的奥秘。这棵树被称为"奇幻之树"，因为它拥有神奇的能力。每个节点都有自己独特的颜色，这些颜色代表了树上不同生物的种类。小蓝注意到，树的每个节点都有一个颜色标签 ci​ ，而这些颜色标签在树上不断变化，使得树上的生物多样性持续增加。他提出了一些问题，希望能够了解奇幻之树上不同颜色的种类数量。每次他会选择一颗树上的节点，然后探索以这个节点为根的**子树**中的颜色种类。

具体问题如下：给定一棵节点为 n 的有根树，**根节点为** 1 ，每个节点有一个颜色 ci ，小蓝有 q 组询问，每次询问给定一个节点 x ，问 x 为根的**子树**中，有多少种不同的颜色。
- 输入格式：第一行输入两个整数 n,q 。代表树的节点数量与询问数量。接下来一行，输入 n 个整数 c1​,c2​,c3​...cn​ ， 代表每个节点的颜色。接下来 n−1 行，每行两个整数 ui​,vi​ ，代表存在一条边连接 ui​,vi​ 两点。接下来 q 行， 每行一个整数 xi​ ，代表询问的点。
- 输出格式：输出 q 行，每行一个整数，代表每次询问的答案。
- 数据范围：1 ≤ n,q ≤105, 1≤ ci​ ≤100, 1≤ ui​,vi​,xi ​≤ n 。输入保证是一棵树。
```cpp
#include<bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10; // 最大节点数
vector<int> g[maxn]; // 邻接表，表示树，二维数组存储邻接表
int prefix[maxn][101]; // prefix[i][col]表示在dfn为下标的数组区间[i,j] 中col颜色的节点数量
int c[maxn]; // 存储每个节点的颜色
int dfn[maxn], tot = 0, idx[maxn], sz[maxn]; // dfn表示节点的dfs编号，tot是dfs序的计数器，idx是dfs序对应的节点，sz是每个子树的大小

void dfs(int x, int fa) // dfs 函数通常是递归函数，它的参数 dfs(node（正在访问的节点）, parent（访问节点的父节点）)表示
{
    dfn[x] = ++tot; // 给当前节点编号
    idx[dfn[x]] = x; // 记录当前节点的DFS编号
    sz[x] = 1; // 当前节点的子树大小为1（自己）
    for(const auto &y : g[x])
    {
        if(y == fa) continue; // 防止回到父节点
        dfs(y, x); // 递归DFS
        sz[x] += sz[y]; // 累加子树大小
    }
}

int main()
{
    int n, q; cin >> n >> q; // 读取节点数n和查询数q
    for(int i = 1; i <= n; i++) cin >> c[i]; // 读取每个节点的颜色
    for(int i = 1; i < n; i++)
    {
        int u, v; cin >> u >> v; // 读取树的边
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0); // 从根节点1开始DFS，节点（也是根节点） 1 的父节点是 0
    // 计算每个节点的颜色前缀和
    for(int i = 1; i <= 100; i++) // 假设颜色编号为1到100
        for(int j = 1; j <= n; j++)
            prefix[j][i] = prefix[j - 1][i] + (int)(c[idx[j]] == i);
    while(q--) // 处理每个查询
    {
        int x; cin >> x; // 读取查询的节点x
        int l = dfn[x], r = l + sz[x] - 1; // 查询节点x的子树在DFS序列中的区间
        int ans = 0;
        for(int i = 1; i <= 100; i++) // 检查每种颜色
            if(prefix[r][i] - prefix[l - 1][i] > 0) ans++; // 如果该颜色在子树区间内出现过
        cout << ans << '\n'; // 输出不同颜色的数量
    } 
    return 0;
}
```
