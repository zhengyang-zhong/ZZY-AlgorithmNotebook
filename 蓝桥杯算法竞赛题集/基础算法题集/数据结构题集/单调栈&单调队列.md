# 最大区间
给定一个长度为 n 的序列 Ai​，求 L,R 使 (R−L+1)⋅min(AL​,AL+1​,…,AR​) 尽可能大，其中 min 表示最小值。你只需要输出最大的值即可，不需要输出具体的 L,R。
- 输入格式：输入的第一行包含一个整数 n。第二行包含 n 个整数，分别表示 A1,A2,…,AnA1​,A2​,…,An​，相邻两个整数之间使用一个空格分隔。
- 输出格式：输出一行包含一个整数表示答案。
```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

void solution()
{
  int n; cin >> n;
  vector<LL> a(n), l(n, -1), r(n, n);
  for(int i = 0; i < n; i++) cin >> a[i];

  deque<int> s;
  // 遍历每个Ai，找到每个Ai对应左侧第一个比自己小的值的位置 
  for(int i = 0; i < n; i++)
  {
    while(s.size() && a[s.back()] >= a[i]) s.pop_back();
    if(s.size()) l[i] = s.back();
    s.push_back(i);
  }

  s.clear();

  // 遍历每个Ai，找到右侧第一个比自己小的值的位置
  for(int i = n - 1; i >= 0; i--)
  {
    while(s.size() && a[s.back()] >= a[i]) s.pop_back();
    if(s.size()) r[i] = s.back();
    s.push_back(i);
  }

  LL ans = 0;
  for(int i = 0; i < n; i++)
    ans = max(ans, 1LL* (r[i] - l[i] - 1) * a[i]);

  cout << ans << '\n';
}


int main()
{
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int t = 1;
  while(t--) solution();
  
  return 0;
}
```

伦太郎特别喜欢喝胡椒博士汽水。他买了 NN 瓶胡椒博士汽水，每一瓶都有一个美味值 ViVi​。但是伦太郎并不是一个简单的吃货，他将 NN 瓶汽水从左到右排成一排放在桌上，编号从 11 开始。他想知道存在多少个这样的“美味区间”，使得在区间内的所有胡椒博士汽水中（美味区间是一个闭区间，区间长度不小于 22），最大的美味值与最小的美味值之差不大于 KK。

第一行两个正整数 N,K，分别表示有 N 瓶胡椒博士汽水和一个美味值的极差上限。第二行 N 个正整数 Vi​，代表第 i 瓶胡椒博士汽水的美味值。

输出共 11 行，输出一个整数表示“美味区间”的个数。

```text
6 3
3 14 5 8 9 6

4
```

```c++
#include<bits/stdc++.h>
using namespace std;

const int maxn = 1e5 + 10;
int val[maxn] = {}, n, k, ans = 0;
deque<int> qmax, qmin;


void test(){
	cin >> n >> k;
	for(int i = 1; i <= n; i++) cin >> val[i];
	for(int left = 1, right = 1; left <= n; left++)
	{
		while(right <= n)
		{
			while(!qmax.empty() && val[qmax.back()] <= val[right]) qmax.pop_back();
			qmax.push_back(right);
			while(!qmin.empty() && val[qmin.back()] >= val[right]) qmin.pop_back();
			qmin.push_back(right);
			if(val[qmax.front()] - val[qmin.front()] > k) break;
			right++;
		}
		if(qmax.front() == left) qmax.pop_front();
		if(qmin.front() == left) qmin.pop_front();
		
		ans += right - left - 1;
	}
	
	cout << ans << endl;
}

int main()
{
	ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
	
	test();
	
    return 0;    
}
```
