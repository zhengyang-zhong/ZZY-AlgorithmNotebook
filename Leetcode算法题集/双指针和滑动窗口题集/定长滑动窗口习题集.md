- [1456. 定长子串中元音的最大数目](https://leetcode.cn/problems/maximum-number-of-vowels-in-a-substring-of-given-length/) 
给你字符串 `s` 和整数 `k` 。请返回字符串 `s` 中长度为 `k` 的单个子字符串中可能包含的最大元音字母数。英文中的 **元音字母** 为（`a`, `e`, `i`, `o`, `u`）。
```c++
class Solution {

public:

    int maxVowels(string s, int k) {

        int maxans = 0, vowel = 0;

        for(int i = 0; i < s.length(); i++)

        {    //1.进入窗口

            //一个新单元进入窗口

            if(s[i] == 'a' || s[i] == 'e' || s[i] == 'i' || s[i] == 'o' || s[i] == 'u')

                vowel++;

            //2.更新答案

            //判断当前窗口大小是否为k，下标从k-1开始统计子串当前总元音个数，不足则先累计，之后等于k只需每更换一个单元比较新子串

            if(i < k - 1)  continue;

            maxans = max(maxans, vowel);

            //3.离开窗口

            //离开窗口的对应单元情况进行统计

            char out = s[i - k + 1];

            if(out == 'a' || out == 'e' || out == 'i' || out == 'o' || out == 'u')

                vowel--;

        }

        return maxans;

    }

};
```

- [643. 子数组最大平均数 I](https://leetcode.cn/problems/maximum-average-subarray-i/)
给你一个由 `n` 个元素组成的整数数组 `nums` 和一个整数 `k` 。请你找出平均数最大且 **长度为 `k`** 的连续子数组，并输出该最大平均数。任何误差小于 `10-5` 的答案都将被视为正确答案。
```c++
class Solution {

public:

    double findMaxAverage(vector<int>& nums, int k) {

        double maxans = INT_MIN, avernum = 0;

        for(int i = 0; i < nums.size(); i++)

        {

            avernum += (double)nums[i];

            if(i < k-1) continue;

            maxans = maxans >= avernum / k ? maxans : avernum / k;

            avernum -= nums[i - k+1];

        }

        return maxans;

    }

};
```
- [2090. 半径为 k 的子数组平均值](https://leetcode.cn/problems/k-radius-subarray-averages/) 
给你一个下标从 **0** 开始的数组 `nums` ，数组中有 `n` 个整数，另给你一个整数 `k` 。
**半径为 k 的子数组平均值** 是指：`nums` 中一个以下标 `i` 为 **中心** 且 **半径** 为 `k` 的子数组中所有元素的平均值，即下标在 `i - k` 和 `i + k` 范围（**含** `i - k` 和 `i + k`）内所有元素的平均值。如果在下标 `i` 前或后不足 `k` 个元素，那么 **半径为 k 的子数组平均值** 是 `-1` 。
构建并返回一个长度为 `n` 的数组 `avgs` ，其中 `avgs[i]` 是以下标 `i` 为中心的子数组的 **半径为 k 的子数组平均值** 。
`x` 个元素的 **平均值** 是 `x` 个元素相加之和除以 `x` ，此时使用截断式 **整数除法** ，即需要去掉结果的小数部分。
- 例如，四个元素 `2`、`3`、`1` 和 `5` 的平均值是 `(2 + 3 + 1 + 5) / 4 = 11 / 4 = 2.75`，截断后得到 `2` 。
![[Leetcode2090例题图.png]]

```cpp
class Solution {
public:
    vector<int> getAverages(vector<int>& nums, int k) {
        int len = nums.size();
        vector<int> argvs(len, -1);
        if(len < 2*k) return argvs; //先判断数组小于直径长的特殊情况，直接返回不判断

        long int sum = 0;
        for(int i = 0; i < len; i++)
        {
            sum +=nums[i];
            if(i >= 2*k) 
            {
                argvs[i-k] = sum / (2*k +1);
                sum -= nums[i-2*k];
            }
            else continue;
        }
        return argvs;
    }
};
```

- [1652. 拆炸弹](https://leetcode.cn/problems/defuse-the-bomb/) （取模循环）
你有一个炸弹需要拆除，时间紧迫！你的情报员会给你一个长度为 n 的 循环 数组 code 以及一个密钥 k 。
为了获得正确的密码，你需要替换掉每一个数字。所有数字会 同时 被替换。
如果 k > 0 ，将第 i 个数字用 接下来 k 个数字之和替换。
如果 k < 0 ，将第 i 个数字用 之前 k 个数字之和替换。
如果 k == 0 ，将第 i 个数字用 0 替换。
由于 code 是循环的， `code[n-1] 下一个元素是 code[0] ，且 code[0] 前一个元素是 code[n-1] `。给你 循环 数组 code 和整数密钥 k ，请你返回解密后的结果来拆除炸弹！

思路：
注意无论 k>0 还是 k<0，窗口都是在向右移动的，所以确定好第一个窗口的位置，就可以把 k>0 和 k<0 两种情况合并起来了。
- k>0，第一个窗口的的下标范围为 `[1,k+1)`。
- k<0，第一个窗口的的下标范围为 `[n−∣k∣,n)`。
无论 k 是正是负，窗口的大小都是 ∣k∣。在窗口向右滑动时，设移入窗口的元素下标为 `r mod n`，则移出窗口的元素下标为 `(r−∣k∣) mod n`。代码实现时，k=0 的特判可以省略。

```cpp
class Solution {
public:
    vector<int> decrypt(vector<int>& code, int k) {
        int len = code.size();
        vector<int> codeans(len, 0);
        int r = k > 0 ? k + 1 : len;
        //k>0，第一个窗口的的下标范围为 [1,k+1)。
        //k<0，第一个窗口的的下标范围为 [n−∣k∣,n)。
        k = k > 0 ? k : -k; //无论 k 是正是负，窗口的大小都是 ∣k∣。
        int sum = 0;
        for(int i = r - k; i < r; i++)
//下标为r的数组单元在下面的循环中开始作为滑入窗口单元计算，当前先计算从下标r-k到r-1这k个窗口大小的数组和
            sum += code[i]; //计算初始的 sum（codeans[0]）值
        for(int i = 0; i < len; i++)
        {
            codeans[i] = sum;
            sum += code[r % len] - code[(r-k) % len]; //滑动窗口一进一出
            r++;
            /*
            code[r % len]：表示 r 位置的元素，使用取模操作来实现循环访问。
            code[(r - k) % len]：表示 r-k 位置的元素，进行减法操作。
            更新 s 的过程是通过从 s 中加上当前索引 r 对应的值，然后减去 r-k 对应的值。
            r++：更新 r，继续处理下一个元素。
            */
        }
        return codeans;
    }
};
```


- [1423. 可获得的最大点数](https://leetcode.cn/problems/maximum-points-you-can-obtain-from-cards/) （逆向思维）
几张卡牌 **排成一行**，每张卡牌都有一个对应的点数。点数由整数数组 `cardPoints` 给出。每次行动，你可以从行的开头或者末尾拿一张卡牌，最终你必须正好拿 `k` 张卡牌。你的点数就是你拿到手中的所有卡牌的点数之和。给你一个整数数组 `cardPoints` 和整数 `k`，请你返回可以获得的最大点数。

![[Leetcode1423例题图.png]]

思路：逆向化求最大为求最小值思维，使用滑动窗口求解
拿走 k 张，剩下 n−k 张。这里 n 是 cardPoints 的长度。由于拿走的点数和 + 剩下的点数和 = 所有点数和 = 常数，所以为了最大化拿走的点数和，应当最小化剩下的点数和。由于只能从开头或末尾拿牌，所以最后剩下的牌必然是连续的。

至此，问题变成：计算长为 n−k 的连续子数组和的最小值。

```cpp
class Solution {
public:
    int maxScore(vector<int> &cardPoints, int k) {
        int len = cardPoints.size();
        int remainlen = len - k;
        int tmpsum = 0;
        for(int i = 0; i < remainlen; i++) tmpsum += cardPoints[i];
        int minsum = tmpsum;
        int totalval = 0;
        for(int i = 0; i < len; i++) totalval += cardPoints[i];
        for(int i = remainlen; i < len; i++)
        {
            // 滑动窗口每向右移动一格，增加从右侧进入窗口的元素值，并减少从左侧离开窗口的元素值
            tmpsum += cardPoints[i] - cardPoints[i - remainlen];
            minsum = minsum < tmpsum ? minsum : tmpsum;
        }
        return totalval - minsum;
    }
};
```

- [1461. 检查一个字符串是否包含所有长度为 K 的二进制子串](https://leetcode.cn/problems/check-if-a-string-contains-all-binary-codes-of-size-k/)
给你一个二进制字符串 `s` 和一个整数 `k` 。如果所有长度为 `k` 的二进制字符串都是 `s` 的子串，请返回 `true` ，否则请返回 `false` 。示例 1：
输入：s = "00110110", k = 2
输出：true
解释：长度为 2 的二进制串包括 "00"，"01"，"10" 和 "11"。它们分别是 s 中下标为 0，1，3，2 开始的长度为 2 的子串。

```cpp
class Solution {
public:
    bool hasAllCodes(string s, int k) {
        if(k > s.size()) return false;
        
        vector<bool> hasharray( 1<<k , 0);   //由规律得出hash表的表长为2的k次方，1 << k 表示 2^k
        int cur = 0;
        for(int i = 0; i < s.size(); i++)
        {
            cur = cur*2 + (s[i] == '1'); //滑动窗口入阶段：累加当前子串长度
            if(i < k-1) continue; //如果没达到k个（对应下标k-1）则继续累加
            hasharray[cur] = 1; //滑动窗口更新阶段：对应hash表下标单元内记录存在此排列
            cur &= ~(1<<(k-1)); //滑动窗口出阶段：将从右往左数的第k个元素按位运算剔除，准备加入下一个单元
        }
        for(int i = 0; i < hasharray.size(); i++)
            if(!hasharray[i]) return false;
        return true;
    }
};
```

注意：
- `1 << k` 的结果是 `2^k`，即 `1` 左移 `k` 位，等同于 `1 * 2^k`，并且它是一个快速的计算方式，用来得到 `2` 的 `k` 次方。
- `2 << k` 的结果是 `2` 乘以 `2^k`，也可以理解为将数字 2 左移 `k` 位，效果是得到一个 `2^k` 的倍数。

- [2841. 几乎唯一子数组的最大和](https://leetcode.cn/problems/maximum-sum-of-almost-unique-subarray/)
给你一个整数数组 `nums` 和两个正整数 `m` 和 `k` 。请你返回 `nums` 中长度为 `k` 的 **几乎唯一** 子数组的 **最大和** ，如果不存在几乎唯一子数组，请你返回 `0` 。如果 `nums` 的一个子数组有至少 `m` 个互不相同的元素，我们称它是 **几乎唯一** 子数组。子数组指的是一个数组中一段连续 **非空** 的元素序列。

示例 2：
输入：nums = [5,9,9,2,4,5,4], m = 1, k = 3
输出：23
解释：总共有 5 个长度为 k = 3 的几乎唯一子数组。分别为 [5, 9, 9] ，[9, 9, 2] ，[9, 2, 4] ，[2, 4, 5] 和 [4, 5, 4] 。这些子数组中，和最大的是 [5, 9, 9] ，和为 23 。

```cpp
class Solution {
public:
    long long maxSum(vector<int>& nums, int m, int k) {
        long long maxsum = 0, tmpsum = 0;
        int cnt = 0;
        unordered_map<int, int> hashcnt;
        
        for(int i = 0; i < nums.size(); i++)
        {
            //入阶段：先积累一个新元素判断是否达到更新标准
            tmpsum += nums[i];
            hashcnt[nums[i]]++;
            if(i < k-1) continue;
            
            //更新阶段：大于等于最小不同元素数时进行滑动窗口子数组总和比大小
            if(hashcnt.size() >= m) 
                maxsum = maxsum > tmpsum ? maxsum : tmpsum;
           
           //出阶段：删除i-k+1下标对应的元素，并查看对应的hash表中对应该元素是否只被统计1次（即要被删除的元素不同于当前数组其他元素），是则在hash表中删除该元素
            tmpsum -= nums[i - k + 1];
            if(--hashcnt[nums[i-k+1]] == 0) 
                hashcnt.erase(nums[i-k+1]); //删除键为nums[i-k+1]的元素
        }

        return maxsum;
    }
};
```

- [2461. 长度为 K 子数组中的最大和](https://leetcode.cn/problems/maximum-sum-of-distinct-subarrays-with-length-k/)
给你一个整数数组 `nums` 和一个整数 `k` 。请你从 `nums` 中满足下述条件的全部子数组中找出最大子数组和：
- 子数组的长度是 `k`，且
- 子数组中的所有元素 **各不相同 。**
返回满足题面要求的最大子数组和。如果不存在子数组满足这些条件，返回 `0` 。**子数组** 是数组中一段连续非空的元素序列。

```cpp
class Solution {
public:
    long long maximumSubarraySum(vector<int>& nums, int k) {
        if(k > nums.size()) return 0;
        
        unordered_map<int, int> hash;

        long long sum = 0, maxsum = 0;
        int cnt = 0; //记录当前组内互异个数
        for(int i = 0; i < nums.size(); i++)
        {
            sum += nums[i];
            hash[nums[i]]++; //增加 hash[nums[i]] 对应的计数，并在hash中更新该元素的出现次数。
            if(hash[nums[i]] == 1) cnt++;
            if(i < k-1) continue; 

            if(cnt == k) maxsum = maxsum > sum ? maxsum : sum;
            
            sum -= nums[i-k+1];
            hash[nums[i-k+1]]--;
            if(!hash[nums[i-k+1]]) cnt--;
        }
        return maxsum;
    }
};
```

- [2134. 最少交换次数来组合所有的 1 II](https://leetcode.cn/problems/minimum-swaps-to-group-all-1s-together-ii/)  （断环成链，数组长度增大一倍）
交换 定义为选中一个数组中的两个 互不相同 的位置并交换二者的值。环形 数组是一个数组，可以认为 第一个 元素和 最后一个 元素 相邻 。给你一个 二进制环形 数组 nums ，返回在 任意位置 将数组中的所有 1 聚集在一起需要的最少交换次数。
```cpp
class Solution {
public:
    int minSwaps(vector<int>& nums) {
        int len = nums.size(), cnt = 0;
        int minSwapstimes = INT_MAX;
        for(int i = 0; i < len; i++)
            if(nums[i]) cnt++;
        
        for(int i = 0; i < len; i++)
            nums.push_back(nums[i]); //将原数组增长为自身原长度的2倍，断环成链才能利用首尾相连的性质
            
        int tmpsum = 0;
        for(int i = 0; i < 2*len; i++)
        {
            tmpsum += nums[i];
            if(i < cnt-1) continue;
            if(tmpsum < cnt) minSwapstimes = minSwapstimes < cnt - tmpsum ? minSwapstimes : cnt - tmpsum;
            else return 0;
            tmpsum -= nums[i - cnt + 1];
        }


        return minSwapstimes;
    }
};
```

- [2653. 滑动子数组的美丽值](https://leetcode.cn/problems/sliding-subarray-beauty/) 
给你一个长度为 `n` 的整数数组 `nums` ，请你求出每个长度为 `k` 的子数组的 **美丽值** 。一个子数组的 **美丽值** 定义为：如果子数组中第 `x` **小整数** 是 **负数** ，那么美丽值为第 `x` 小的数，否则美丽值为 `0` 。请你返回一个包含 `n - k + 1` 个整数的数组，**依次** 表示数组中从第一个下标开始，每个长度为 `k` 的子数组的 **美丽值** 。子数组指的是数组中一段连续 **非空** 的元素序列。数据说明： `-50 <= nums[i] <= 50`
```cpp
class Solution {
public:
//第 x 小的数是指在一系列数中按照大小顺序排序后，排在第 x 个位置的数。
    vector<int> getSubarrayBeauty(vector<int>& nums, int k, int x) {
        vector<int> ans(nums.size() - k +1, 0);
        const int bias = 50; //由于数的范围是在[-50, 50]之间，所以之后要利用这些数做下标时先变成非负数
        int cnt[bias * 2 + 1]{};
        for(int i = 0; i < nums.size(); i++)
        {
            cnt[nums[i] + bias] ++; //利用下标和偏移量转成正数来记录数组元素对应出现的次数（因为可能有相同数字出现重复）
            if(i < k - 1) continue;
            
            int left = x; //左侧小于x的次数
            for(int j = 0; j < bias; j++) //参考计数排序的做法找到第x小的数，暴力枚举负数范围 [-50,-1]
            {
                left -= cnt[j]; //减去累计次数
                if(left <= 0) //符合条件时说明j已经是第x小的数
                {
                    ans[i - k + 1] = j - bias; //还原并记录原具体第x小的数
                    break;
                }
            }
            
            cnt[nums[i - k + 1] + bias]--;
        }
        return ans;
    }
};
```

- [1888. 使二进制字符串字符交替的最少反转次数](https://leetcode.cn/problems/minimum-number-of-flips-to-make-the-binary-string-alternating/) （注意长度倍增）
给你一个二进制字符串 s 。你可以按任意顺序执行以下两种操作任意次：
- 类型 1 ：删除 字符串 s 的第一个字符并将它 添加 到字符串结尾。
- 类型 2 ：选择 字符串 s 中任意一个字符并将该字符 反转 ，也就是如果值为 '0' ，则反转得到 '1' ，反之亦然。
请你返回使 s 变成 交替 字符串的前提下， 类型 2 的 最少 操作次数 。我们称一个字符串是 交替 的，需要满足任意相邻字符都不同。比方说，字符串 "010" 和 "1010" 都是交替的，但是字符串 "0100" 不是。

```cpp
class Solution {
public:
    int minFlips(string s) {
       int len = s.size(), ans = len;
       string standard = "01";
       s += s;
       //将s延长自身的一倍是为了模拟操作1，把s的长度n作为遍历区间，当遍历的区间右端点下标走到n+i时，区间长度为n，左端点就是i + 1，这时候区间[i + 1, n + i]可认为是对前i个数进行操作1获得的。
       int samecnt = 0;
       for(int i = 0; i < 2*len; i++)
       {
            //入：判断当前和标准交错字符对应位置元素是否相同，是则加一
            if(s[i] == standard[i%2]) samecnt++;
            if(i < len-1) continue;

            //更新：及时更新当前的最小值
            int tmpmin = samecnt < len - samecnt ? samecnt : len - samecnt;
            ans = ans < tmpmin ? ans : tmpmin;
            //取 ans，01 模式，10 这三个模式需要交换的 samecnt 的最小值，samecnt表示s和01010...这个串相同的字符个数；len - samecnt 表示s和1010...这个串相同的个数。因为0101和1010两者1的个数之和是n, 如果s和0101有相同的地方,和1010必不相同。两者取小值, 这个当前的较小值就是操作二需要反转的个数。最后及时和最终答案比较是否更小，及时更新最小值。

            //出：符合条件的元素离开窗口则及时更新减少对应翻转次数
            if(s[i-len+1] == standard[(i-len+1)%2]) samecnt--;
       }
       return ans;
    }
};
```

- [1297. 子串的最大出现次数](https://leetcode.cn/problems/maximum-number-of-occurrences-of-a-substring/) （**重点：使用子串substr（）结合hash表记录符合条件的子串出现次数**）
给你一个字符串 `s` ，请你返回满足以下条件且出现次数最大的 **任意** 子串的出现次数：
- 子串中不同字母的数目必须小于等于 `maxLetters` 。
- 子串的长度必须大于等于 `minSize` 且小于等于 `maxSize` 。
输入：s = "aababcaab", maxLetters = 2, minSize = 3, maxSize = 4
输出：2
解释：子串 "aab" 在原字符串中出现了 2 次。
它满足所有的要求：2 个不同的字母，长度为 3 （在 minSize 和 maxSize 范围内）。

```cpp
class Solution {
public:
    int maxFreq(string s, int maxLetters, int minSize, int maxSize) {
        int ans = 0, len = s.size();
        unordered_map<string, int> hash;
        
        for(int tail = 0; tail < len - minSize + 1; tail++)
        {
            string tmpstr = s.substr(tail, minSize);
            unordered_map<char, int> check;
            for(int i = 0; i < minSize; i++) check[tmpstr[i]]++;
            /*string str = s.substr(i, minSize);
            unordered_set<char> exist(str.begin(), str.end());或用unordered_set实现*/

            if(check.size() <= maxLetters)
            {
                hash[tmpstr]++;
                ans = ans > hash[tmpstr] ? ans : hash[tmpstr];
            }
        }

        return ans;
    }
};
```

- [3208. 交替组 II](https://leetcode.cn/problems/alternating-groups-ii/)
给你一个整数数组 colors 和一个整数 k ，colors表示一个由红色和蓝色瓷砖组成的环，第 i 块瓷砖的颜色为 colors[i] ：colors[i] == 0 表示第 i 块瓷砖的颜色是 红色 。colors[i] == 1 表示第 i 块瓷砖的颜色是 蓝色 。环中连续 k 块瓷砖的颜色如果是 交替 颜色（也就是说除了第一块和最后一块瓷砖以外，中间瓷砖的颜色与它 左边 和 右边 的颜色都不同），那么它被称为一个 交替 组。
请你返回 交替 组的数目。注意 ，由于 colors 表示一个 环 ，第一块 瓷砖和 最后一块 瓷砖是相邻的。

思路：先增长一倍原数组（断环成链）方便后续从原数组的最后一个元素开始检查最后一个以其开头的交替组是否符合要求，之后按照灵神模板：入——记录交替的配对数：每个01或10为一个配对；更新——符合交替组的数组配对总数为k-1，若当前组符合则ans+1；出——删除交替组头部配对数。循环结束返回ans答案。

```cpp
class Solution {
public:
    int numberOfAlternatingGroups(vector<int>& colors, int k) {
        int ans = 0, len = colors.size(), cntpair = 0;
        for(int i = 0; i < len; i++) colors.push_back(colors[i]);
        for(int tail = 1; tail < len + k - 1; tail++)
        {
            cntpair += colors[tail] != colors[tail-1] ? 1 : 0;
            if(tail < k - 1) continue;
            if(k - cntpair == 1) ans++;
            cntpair -= colors[tail - k + 1] != colors[tail - k + 2] ? 1 : 0;
        }
        return ans;
    }
};
```

- [219. 存在重复元素 II](https://leetcode.cn/problems/contains-duplicate-ii/)
给你一个整数数组 `nums` 和一个整数 `k` ，判断数组中是否存在两个 **不同的索引** `i` 和 `j` ，满足 `nums[i] == nums[j]` 且 `abs(i - j) <= k` 。如果存在，返回 `true` ；否则，返回 `false` 。
```cpp
class Solution {
public:
    bool containsNearbyDuplicate(vector<int>& nums, int k) {
        unordered_set<int> hash;
        for(int i = 0; i < nums.size(); i++)
        {
            auto check = hash.find(nums[i]);
            if(check != hash.end()) return true;

            hash.emplace(nums[i]);
            if(i - k >= 0) hash.erase(nums[i - k]);
        }

        return false;
    }
};
```


- [220. 存在重复元素 III](https://leetcode.cn/problems/contains-duplicate-iii/) （**重点：利用set有序集合的不同于定长滑窗模板的写法**）
给你一个整数数组 nums 和两个整数 indexDiff 和 valueDiff 。找出满足下述条件的下标对 (i, j)：`i != j`，`abs(i - j) <= indexDiff`，`abs(nums[i] - nums[j]) <= valueDiff`。如果存在，返回 true ；否则返回 false 。
```cpp
class Solution {
public:
    bool containsNearbyAlmostDuplicate(vector<int>& nums, int indexDiff, int valueDiff) {
        set<long> check;
        for(int i = 0; i < nums.size(); i++)
        {            
            auto tmpval = check.lower_bound(nums[i] - valueDiff);
            if(tmpval != check.end() && *tmpval <= (long) nums[i] + valueDiff) return true;//采用long是为了避免潜在的int类型溢出风险
            //因为要二分查找，为了避免查找到自己，所以要先进行判断，再添加，再收缩（留一个位置让下一轮进行判断） 不同于之前的滑动窗口模板
            //lower_bound(x) 返回的是一个指向第一个大于或等于 x 的元素的迭代器。
            check.emplace(nums[i]);
            if(i - indexDiff >= 0) check.erase(nums[i - indexDiff]); //当达到indexDiff下标值时开始删去定长滑窗的头元素，这里的定长是最大上限值，取决于indexDiff，若还未到达窗口最大值就找到符合的匹配选项则立刻返回true，也没必要再进行至此步。

        }
        return false;
    }
};
```
